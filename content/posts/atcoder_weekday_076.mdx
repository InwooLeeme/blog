};

요즘 들어서 AWC에서 보이는 세그 + 오프라인 쿼리 문제인데 그냥 기본 난이도라 쉽게 풀린다.

```cpp showLineNumbers
struct segtree{
	const int sz = 1 << 17;
	vector<int> tree;
	segtree():tree(sz << 1){}
	void update(int i, int val){
		--i |= sz; tree[i] += val;
		while(i >>= 1) tree[i] = tree[i << 1] + tree[i << 1 | 1];
	}
	int query(int l, int r){
		int ret = 0;
		--l |= sz; --r |= sz;
		while(l <= r){
			if(l & 1) ret += tree[l++];
			if(~r & 1) ret += tree[r--];
			l >>= 1, r >>= 1;
		}
		return ret;
	}
};

int n,k, q;

void Main(){
	ri(n, k, q);
	vector<array<int, 2>> D;
	for(int i = 1; i <= n; i++){
		int x; ri(x); D.push_back({x, i});
	}
	sort(D.begin(), D.end());
	segtree seg;
	vector<array<int, 4>> query;
	for(int i = 0; i < q; i++){
		int a,b,c; ri(a, b, c);
		query.push_back({a, b, c, i});
	}
	sort(query.begin(), query.end(), [&](auto& a, auto& b){
		return a[2] < b[2];
	});
	int idx = 0;
	vector<int> res(q);
	for(const auto& [l, r, t, I] : query){
		while(idx < n and D[idx][0] <= t){
			seg.update(D[idx][1], 1);
			idx++;
		}
		int s = seg.query(l, r);
		res[I] = (s <= k ? s : -1);
	}
	for(const auto& i : res) po(i);
}
```
